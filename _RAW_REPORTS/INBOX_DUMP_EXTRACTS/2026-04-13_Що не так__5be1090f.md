def translate_batch(rotator, system_prompt, cards):
    cards_min = []
    for c in cards:
        card = {"code": c["code"], "name": c["name"],
                "type_name": c.get("type_name",""), "faction_name": c.get("faction_name","")}
        for f in ["subname","traits","slot","text","flavor","back_text","back_flavor","encounter_name"]:
            if c.get(f): card[f] = c[f]
        cards_min.append(card)

    user_msg = json.dumps(cards_min, ensure_ascii=False, indent=2)
    backoff = 5

    for attempt in range(3 * len(rotator.keys) + 5):
        try:
            client = rotator.get_client()
            response = client.chat.completions.create(
                model=MODEL,
                messages=[
                    {"role": "system", "content": system_prompt},
                    {"role": "user", "content": user_msg},
                ],
                temperature=0.3,
                max_tokens=16000,
            )

            text = response.choices[0].message.content.strip()
            if text.startswith("```"):
                text = text.split("\n", 1)[1] if "\n" in text else text[3:]
            if text.endswith("```"):
                text = text[:-3]
            text = text.strip()
            result = json.loads(text)

            if len(result) != len(cards):
                print(f" ⚠{len(result)}≠{len(cards)}", end="", flush=True)
                time.sleep(2)
                continue

            backoff = 5  # reset on success
            return result

        except json.JSONDecodeError:
            print(" ⚠json", end="", flush=True)
            time.sleep(2)
        except Exception as e:
            err = str(e)
            if "429" in err or "rate" in err.lower():
                has_more = rotator.rotate()
                if not has_more:
                    print(f" ⏳{backoff}s", end="", flush=True)
                    time.sleep(backoff)
                    backoff = min(backoff * 2, 300)
                time.sleep(1)
            elif "503" in err or "502" in err or "overloaded" in err.lower():
                print(f" ⏳{backoff}s", end="", flush=True)
                time.sleep(backoff)
                backoff = min(backoff * 2, 300)
                rotator.rotate()
            elif "403" in err or "401" in err:
                print(f" ⚠auth", end="", flush=True)
                rotator.rotate()
                time.sleep(1)
            else:
                print(f" ⚠{err[:50]}", end="", flush=True)
                time.sleep(5)
    return None
